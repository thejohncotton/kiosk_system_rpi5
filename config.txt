# VHSynth Nerves RPi5 config.txt

# Disable the boot rainbow
disable_splash=1

# GPU memory split
gpu_mem=192

# Enable I2C and SPI
dtparam=i2c_arm=on
dtparam=spi=on

# Enable DRM VC4 V3D driver for HDMI output
dtoverlay=vc4-kms-v3d
max_framebuffers=2

# Raspberry Pi Touch Display 2 on DISP/CAM 0 (22-pin connector closer to USB-C)
dtoverlay=vc4-kms-dsi-7inch-v2


# Don't have the firmware create an initial video= setting in cmdline.txt.
disable_fw_kms_setup=1

# Force HDMI hotplug detection on both ports regardless of HPD signal
hdmi_force_hotplug=1

# Run in 64-bit mode
arm_64bit=1

# Disable compensation for displays with overscan
disable_overscan=1

# Run as fast as firmware / board allows
arm_boost=1

# USB gadget mode (used for g_ether on dwc2 OTG port)
dtoverlay=dwc2

# Preserve crash information across reboots in DRAM
dtoverlay=ramoops

# Enable the UART (/dev/ttyAMA0) for serial console
enable_uart=1
